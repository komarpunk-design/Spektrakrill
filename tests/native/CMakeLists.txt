cmake_minimum_required(VERSION 3.22)
project(spektrafilm_experimental_host CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../android/engine/spektra-core/src/main/cpp" ABSOLUTE)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS ${ROOT}/kernels/*.cpp ${ROOT}/model/*.cpp ${ROOT}/profiles/*.cpp ${ROOT}/runtime/*.cpp ${ROOT}/io/*.cpp ${ROOT}/gpu/*.cpp ${ROOT}/ml/*.cpp)
find_package(Threads REQUIRED)
add_library(spektra_host SHARED ${ROOT}/spektra.cpp ${SOURCES})
target_include_directories(spektra_host PRIVATE ${ROOT})
target_compile_options(spektra_host PRIVATE -O2)
target_link_libraries(spektra_host PRIVATE Threads::Threads)
add_executable(experimental_probe experimental_probe.cpp)
target_include_directories(experimental_probe PRIVATE ${ROOT})
target_link_libraries(experimental_probe PRIVATE spektra_host)
add_executable(negs_probe negs_probe.cpp)
target_include_directories(negs_probe PRIVATE ${ROOT})
target_link_libraries(negs_probe PRIVATE spektra_host)
add_executable(output_probe output_probe.cpp)
target_include_directories(output_probe PRIVATE ${ROOT})
target_link_libraries(output_probe PRIVATE spektra_host)
add_executable(transport_probe transport_probe.cpp)
target_include_directories(transport_probe PRIVATE ${ROOT})
target_link_libraries(transport_probe PRIVATE spektra_host)

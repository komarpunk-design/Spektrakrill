cmake_minimum_required(VERSION 3.22)
project(spektrafilm_chat LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CORE "${REPO}/native-android/engine/spektra-core/src/main/cpp")
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "${CORE}/kernels/*.cpp" "${CORE}/model/*.cpp" "${CORE}/profiles/*.cpp" "${CORE}/runtime/*.cpp" "${CORE}/io/*.cpp" "${CORE}/gpu/*.cpp" "${CORE}/ml/*.cpp")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/tests/")
find_package(Threads REQUIRED)
add_library(chat_spektra_core SHARED "${CORE}/spektra.cpp" ${CORE_SOURCES})
target_include_directories(chat_spektra_core PUBLIC "${CORE}")
target_compile_options(chat_spektra_core PRIVATE -O2)
target_link_libraries(chat_spektra_core PRIVATE Threads::Threads)
set_source_files_properties("${CORE}/model/noritsu_operators.cpp" "${CORE}/model/noritsu_advanced.cpp" PROPERTIES COMPILE_OPTIONS "-fno-fast-math;-ffp-contract=off")
# Same pinned, reviewed LibRaw source/patch resolver as the Android decoder.
set(SFRAW_HOST_ENABLE_SANITIZERS OFF CACHE BOOL "" FORCE)
set(SFRAW_HOST_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/native-android/lib/libraw/src/test/host" raw EXCLUDE_FROM_ALL)
set(RAW "${REPO}/native-android/lib/libraw/src/main/cpp")
add_executable(spektrafilm-develop develop.cpp "${RAW}/raw_decoder.cpp")
target_include_directories(spektrafilm-develop PRIVATE "${RAW}")
target_compile_definitions(spektrafilm-develop PRIVATE USE_JPEG=1 USE_ZLIB=1)
target_link_libraries(spektrafilm-develop PRIVATE chat_spektra_core sfraw_host_libraw)
